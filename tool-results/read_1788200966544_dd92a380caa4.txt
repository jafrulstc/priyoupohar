    76→/* ---------------------------- rolling number ---------------------------- */
    77→
    78→function RollingNumber({ value, className }: { value: number; className?: string }) {
    79→  const raw = useMotionValue(value);
    80→  const spring = useSpring(raw, { stiffness: 90, damping: 20 });
    81→  const display = useTransform(spring, (v: number) => formatINR(Math.round(v)));
    82→
    83→  useEffect(() => {
    84→    raw.set(value);
    85→  }, [raw, value]);
    86→
    87→  return <motion.span className={className}>{display}</motion.span>;
    88→}
