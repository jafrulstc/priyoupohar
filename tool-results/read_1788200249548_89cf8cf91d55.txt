    63→
    64→export default function CartDrawer() {
    65→  const isOpen = useShopStore((s) => s.isCartOpen);
    66→  const setOpen = useShopStore((s) => s.setCartOpen);
    67→  const cart = useShopStore((s) => s.cart);
    68→  const updateQty = useShopStore((s) => s.updateQty);
    69→  const removeFromCart = useShopStore((s) => s.removeFromCart);
    70→  const clearCart = useShopStore((s) => s.clearCart);
    71→  const location = useShopStore((s) => s.location);
    72→  const setLocationOpen = useShopStore((s) => s.setLocationOpen);
    73→  const setLastOrderId = useShopStore((s) => s.setLastOrderId);
    74→  const setTrackOpen = useShopStore((s) => s.setTrackOpen);
    75→  /* upsells + loyalty */
    76→  const premiumWrap = useShopStore((s) => s.premiumWrap);
    77→  const setPremiumWrap = useShopStore((s) => s.setPremiumWrap);
    78→  const deliverySlot = useShopStore((s) => s.deliverySlot);
    79→  const setDeliverySlot = useShopStore((s) => s.setDeliverySlot);
    80→  const stamps = useShopStore((s) => s.stamps);
    81→  const rewardCoupon = useShopStore((s) => s.rewardCoupon);
    82→  const recordOrder = useShopStore((s) => s.recordOrder);
    83→  const dismissReward = useShopStore((s) => s.dismissReward);
    84→  const { toast } = useToast();
    85→
    86→  const mounted = useMounted();
    87→  const [checkingOut, setCheckingOut] = useState(false);
    88→  const [result, setResult] = useState<CheckoutResult | null>(null);
    89→  const [unlocked, setUnlocked] = useState<string | null>(null);
    90→  const [couponInput, setCouponInput] = useState("");
    91→  const [coupon, setCoupon] = useState<string | null>(null);
    92→  const [couponError, setCouponError] = useState<string | null>(null);
    93→  const appliedCoupon = resolveCoupon(coupon);
    94→
    95→  // Free-shipping crossing tracker

    97→

   107→      window.removeEventListener("keydown", onKey);
   108→    };

   110→
   111→  const subtotal = mounted ? cartTotal(cart) : 0;

   117→  const deliveryFee = pct >= 100 || appliedCoupon?.kind === "shipping" ? 0 : 99;

   122→    const code = couponInput.trim().toUpperCase();

   133→      });

   141→      prevPct.current = 0;

   150→
   151→  const checkout = async () => {
   152→    if (cart.length === 0) return;
   153→    if (!location) {
   154→      toast({
   155→        title: "Where are we heading? 📍",
   156→        description: "Pick a delivery city first.",
   157→      });

   159→      setLocationOpen(true);
   160→      return;
   161→    }

   164→      const res = await fetch("/api/checkout", {
   165→        method: "POST",
   166→        headers: { "Content-Type": "application/json" },
   167→        body: JSON.stringify({
   168→          items: cart.map((c) => ({ id: c.id, name: c.name, price: c.price, qty: c.qty })),
   169→          location: { city: location.city, pincode: location.pincode },
   170→          slot: deliverySlot,

   172→          premiumWrap,

   181→        id: data.orderId,
   182→        total: data.total,

   194→      celebrationConfetti();
   195→    } catch {
   196→      toast({ title: "Checkout hiccup 😢", description: "Please try again in a moment." });

   198→      setCheckingOut(false);
   199→    }
   200→  };
   201→
   202→  const finish = () => {

   204→    setResult(null);
   205→    setUnlocked(null);
   206→    setPremiumWrap(false);
   207→    setCoupon(null);
   208→    setCouponInput("");
   209→    setCouponError(null);
   210→    setOpen(false);
   211→  };

   220→            animate={{ opacity: 1 }}
   221→            exit={{ opacity: 0 }}
   222→            transition={{ duration: 0.2 }}

   225→          />

   227→          {/* Panel */}

   229→            initial={{ x: "100%" }}

   233→            className="absolute inset-y-0 right-0 flex w-full max-w-md flex-col bg-cream shadow-2xl"

   236→            <div className="flex items-center justify-between border-b border-rose-100 bg-white px-5 py-4">

   239→                  <ShoppingBag className="h-4.5 w-4.5" aria-hidden />
   240→                </span>
   241→                <div>
   242→                  <h2 className="text-base font-extrabold text-charcoal">Your Gift Bag</h2>
   243→                  <p className="text-xs text-stone-400">{count} item{count === 1 ? "" : "s"} of joy</p>
   244→                </div>
   245→              </div>
   246→              <button
   247→                onClick={() => setOpen(false)}
   248→                className="grid h-9 w-9 place-items-center rounded-full border border-stone-200 text-stone-500 transition hover:border-rose-300 hover:text-brand"
   249→                aria-label="Close gift bag"
   250→              >
   251→                <X className="h-4.5 w-4.5" />
   252→              </button>
   253→            </div>
   254→
   255→            {result ? (
   256→              /* ---------- SUCCESS VIEW ---------- */
   257→              <div className="flex flex-1 flex-col items-center justify-center overflow-y-auto px-6 text-center scrollbar-slim">
   258→                <motion.div
   259→                  initial={{ scale: 0.6, opacity: 0 }}
   260→                  animate={{ scale: 1, opacity: 1 }}
   261→                  transition={{ type: "spring", stiffness: 260, damping: 20 }}

   264→                </motion.div>
   265→                <motion.div
   266→                  initial={{ y: 24, opacity: 0 }}
   267→                  animate={{ y: 0, opacity: 1 }}
   268→                  transition={{ delay: 0.15 }}
   269→                >
   270→                  <h3 className="text-xl font-extrabold text-charcoal">Order confirmed! 🎉</h3>
   271→                  <p className="mt-1 text-sm text-stone-500">
   272→                    Order <span className="font-mono font-bold text-brand">{result.orderId}</span> is

   277→                      <motion.div
   278→                        initial={{ opacity: 0, scale: 0.9 }}

   290→                    <div className="flex justify-between">

   294→                    {result.coupon && result.discount > 0 && (
   295→                      <div className="flex justify-between">
   296→                        <span className="flex items-center gap-1 text-mint">
   297→                          <BadgeCheck className="h-3.5 w-3.5" aria-hidden />
   298→                          {result.coupon} saved
   299→                        </span>

   301→                      </div>
   302→                    )}
   303→                    {result.coupon && result.freeShipping && (
   304→                      <div className="flex justify-between">
   305→                        <span className="flex items-center gap-1 text-mint">
   306→                          <BadgeCheck className="h-3.5 w-3.5" aria-hidden />
   307→                          {result.coupon} · free shipping
   308→                        </span>
   309→                        <span className="font-bold text-mint">−{formatINR(99)}</span>
   310→                      </div>
   311→                    )}
   312→                    <div className="flex justify-between">
   313→                      <span className="text-stone-500">Delivering to</span>
   314→                      <span className="font-bold text-charcoal">{result.deliveryTo}</span>
   315→                    </div>
   316→                    <div className="flex justify-between">
   317→                      <span className="text-stone-500">Slot</span>

   338→                      Keep shopping <ArrowRight className="h-4 w-4" aria-hidden />
   339→                    </button>

   347→                      className="flex items-center justify-center gap-1.5 rounded-full border border-rose-200 bg-white px-4 py-3 text-sm font-bold text-charcoal transition hover:border-brand hover:text-brand"
   348→                    >

   350→                      Track

   353→                </motion.div>
   354→              </div>
   355→            ) : cart.length === 0 ? (
   356→              /* ---------- EMPTY VIEW ---------- */
   357→              <div className="flex flex-1 flex-col items-center justify-center gap-3 px-6 text-center">
   358→                <motion.div
   359→                  animate={{ y: [0, -10, 0], rotate: [0, -4, 4, 0] }}
   360→                  transition={{ repeat: Infinity, duration: 3.2 }}
   361→                  className="grid h-24 w-24 place-items-center rounded-full bg-brand-soft text-brand"
   362→                >
   363→                  <Gift className="h-10 w-10" aria-hidden />
   364→                </motion.div>
   365→                <h3 className="text-lg font-extrabold text-charcoal">Your bag is feeling light</h3>
   366→                <p className="max-w-60 text-sm text-stone-500">
   367→                  Fill it with flowers, cakes & surprises — happiness ships free over{" "}
   368→                  {formatINR(FREE_SHIPPING_THRESHOLD)}.
   369→                </p>
   370→                <button
   371→                  onClick={() => setOpen(false)}
   372→                  className="mt-2 rounded-full bg-brand px-6 py-3 text-sm font-bold text-white shadow-lift transition hover:bg-rose-700"
   373→                >
   374→                  Browse bestsellers
   375→                </button>
   376→              </div>
   377→            ) : (
   378→              <>
   379→                {/* ---------- BLOOM REWARDS ---------- */}
   380→                <div className="border-b border-rose-100 bg-white px-5 py-3">
   381→                  <div className="flex items-center justify-between gap-3 rounded-2xl bg-gradient-to-r from-rose-50 via-pink-50 to-amber-50 px-3.5 py-2.5 ring-1 ring-rose-100">
   382→                    <div className="flex items-center gap-2.5">
   383→                      <motion.span
   384→                        animate={{ rotate: [0, -8, 8, 0] }}
   385→                        transition={{ repeat: Infinity, duration: 3.4, ease: "easeInOut" }}
   386→                        className="grid h-8 w-8 shrink-0 place-items-center rounded-xl bg-brand text-white shadow-soft"
   387→                        aria-hidden
   388→                      >
   389→                        <Flower2 className="h-4 w-4" />
   390→                      </motion.span>
   391→                      <div>

   393→                          Bloom Rewards{" "}
   394→                          <span className="ml-0.5 rounded-full bg-brand-soft px-1.5 py-0.5 text-[9px] font-extrabold uppercase tracking-wide text-brand">
   395→                            {stamps}/{LOYALTY_TARGET} stamps

   397→                        </p>
   398→                        <p className="text-[10px] font-semibold text-stone-400">
   399→                          {stamps > 0
   400→                            ? `${LOYALTY_TARGET - stamps} more order${LOYALTY_TARGET - stamps === 1 ? "" : "s"} → ₹100 off`
   401→                            : "Every 3rd order earns ₹100 off"}
   402→                        </p>
   403→                      </div>

   407→                        <motion.span
   408→                          key={i}
   409→                          initial={false}
   410→                          animate={
   411→                            i < stamps
   412→                              ? { scale: [1, 1.35, 1], rotate: [0, 18, 0] }
   413→                              : { scale: 1 }
   414→                          }
   415→                          transition={{ duration: 0.45, delay: i * 0.12 }}
   416→                        >
   417→                          <Flower2
   418→                            className={cn(
   419→                              "h-4.5 w-4.5",
   420→                              i < stamps ? "fill-brand text-brand" : "text-rose-200"
   421→                            )}
   422→                          />

   426→                  </div>
   427→                  {rewardCoupon && !coupon && (
   428→                    <motion.div
   429→                      initial={{ opacity: 0, y: 6 }}
   430→                      animate={{ opacity: 1, y: 0 }}
   431→                      className="mt-2 flex items-center justify-between rounded-2xl border border-dashed border-gold bg-gold-soft/70 px-3.5 py-2"
   432→                    >
   433→                      <span className="flex items-center gap-1.5 text-[11px] font-extrabold text-amber-700">

   435→                        Your ₹100 reward · {rewardCoupon}

   439→                          onClick={() => {
   440→                            setCoupon(rewardCoupon);
   441→                            setCouponInput(rewardCoupon);
   442→                            miniConfetti();
   443→                            toast({
   444→                              title: "Reward applied! 🌸",
   445→                              description: "BLOOM100 · ₹100 off this order.",
   446→                            });
   447→                          }}
   448→                          className="rounded-full bg-charcoal px-3 py-1 text-[10px] font-extrabold text-cream transition hover:bg-stone-700"

   450→                          Apply

   455→                          className="text-stone-400 transition hover:text-brand"

   463→
   464→                {/* ---------- FREE SHIPPING PROGRESS ---------- */}

   470→                    />
   471→                    {pct >= 100 ? (

   489→                      animate={{ width: `${pct}%` }}
