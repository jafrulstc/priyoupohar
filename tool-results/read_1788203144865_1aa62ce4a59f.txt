    66→export default function CartDrawer() {
    67→  const isOpen = useShopStore((s) => s.isCartOpen);
    68→  const setOpen = useShopStore((s) => s.setCartOpen);
    69→  const cart = useShopStore((s) => s.cart);
    70→  const updateQty = useShopStore((s) => s.updateQty);
    71→  const removeFromCart = useShopStore((s) => s.removeFromCart);
    72→  const clearCart = useShopStore((s) => s.clearCart);
    73→  const location = useShopStore((s) => s.location);
    74→  const setLocationOpen = useShopStore((s) => s.setLocationOpen);
    75→  const setLastOrderId = useShopStore((s) => s.setLastOrderId);
    76→  const setTrackOpen = useShopStore((s) => s.setTrackOpen);
    77→  /* upsells + loyalty */
    78→  const premiumWrap = useShopStore((s) => s.premiumWrap);
    79→  const setPremiumWrap = useShopStore((s) => s.setPremiumWrap);
    80→  const deliverySlot = useShopStore((s) => s.deliverySlot);
    81→  const setDeliverySlot = useShopStore((s) => s.setDeliverySlot);
    82→  const stamps = useShopStore((s) => s.stamps);
    83→  const rewardCoupon = useShopStore((s) => s.rewardCoupon);
    84→  const recordOrder = useShopStore((s) => s.recordOrder);
    85→  const dismissReward = useShopStore((s) => s.dismissReward);
    86→  const { toast } = useToast();
    87→  const isMobile = useIsMobile();
    88→  const trapRef = useFocusTrap<HTMLDivElement>(isOpen);
    89→
    90→  const mounted = useMounted();
    91→  const [checkingOut, setCheckingOut] = useState(false);
    92→  const [result, setResult] = useState<CheckoutResult | null>(null);
    93→  const [unlocked, setUnlocked] = useState<string | null>(null);
    94→  const [couponInput, setCouponInput] = useState("");
    95→  const [coupon, setCoupon] = useState<string | null>(null);
    96→  const [couponError, setCouponError] = useState<string | null>(null);
    97→  const appliedCoupon = resolveCoupon(coupon);
    98→
    99→  // Free-shipping crossing tracker
   100→  const prevPct = useRef(0);
   101→
   102→  // Lock body scroll while open
   103→  useEffect(() => {
   104→    if (!isOpen) return;
   105→    const original = document.body.style.overflow;
   106→    document.body.style.overflow = "hidden";

   108→    window.addEventListener("keydown", onKey);
   109→    return () => {
   110→      document.body.style.overflow = original;
   111→      window.removeEventListener("keydown", onKey);
   112→    };
   113→  }, [isOpen, setOpen]);
   114→
   115→  const subtotal = mounted ? cartTotal(cart) : 0;
   116→  const count = mounted ? cartCount(cart) : 0;
   117→  const pct = Math.min(100, Math.round((subtotal / FREE_SHIPPING_THRESHOLD) * 100));
   118→  const remaining = FREE_SHIPPING_THRESHOLD - subtotal;
   119→  const freeShipUnlocked = pct >= 100 || appliedCoupon?.kind === "shipping";
   120→  const discount = appliedCoupon ? couponDiscount(appliedCoupon, subtotal) : 0;
   121→  const deliveryFee = pct >= 100 || appliedCoupon?.kind === "shipping" ? 0 : 99;
   122→  const wrapFee = premiumWrap ? 49 : 0;
   123→  const grandTotal = subtotal - discount + deliveryFee + wrapFee;
   124→
   125→  /* tiered loyalty: the reward ladder climbs each completed 3-order cycle */
   126→  const ordersCount = useShopStore((s) => s.ordersCount);
   127→  const nextCycle = Math.floor(ordersCount / LOYALTY_TARGET) + 1;
   128→  const nextRewardLabel =
   129→    resolveCoupon(loyaltyRewardFor(nextCycle))?.label ?? "₹100 off";
   130→

   132→    const code = couponInput.trim().toUpperCase();

   134→    if (matched) {
   135→      setCoupon(matched.code);
   136→      setCouponError(null);
   137→      toast({
   138→        title: `${matched.code} applied! 🎊`,
   139→        description:
   140→          matched.kind === "shipping"
   141→            ? "Your gifts ride for free."
   142→            : `${matched.label} your gifts — nicely spotted.`,
   143→      });
   144→    } else {
   145→      setCouponError("That code didn’t work — try BLISS10");
   146→    }
   147→  };
   148→
   149→  useEffect(() => {
   150→    if (!isOpen) {
   151→      prevPct.current = 0;
   152→      return;
   153→    }
   154→    if (pct >= 100 && prevPct.current < 100 && prevPct.current > 0) {
   155→      petalConfetti();
   156→      toast({ title: "FREE shipping unlocked! 🎉", description: "Your gifts ride for free." });
   157→    }
   158→    prevPct.current = pct;
   159→  }, [pct, isOpen, toast]);
   160→
   161→  const checkout = async () => {
   162→    if (cart.length === 0) return;
   163→    if (!location) {
   164→      toast({
   165→        title: "Where are we heading? 📍",
   166→        description: "Pick a delivery city first.",
   167→      });
   168→      setOpen(false);
   169→      setLocationOpen(true);
   170→      return;
   171→    }

   174→      const res = await fetch("/api/checkout", {
   175→        method: "POST",
   176→        headers: { "Content-Type": "application/json" },
   177→        body: JSON.stringify({
   178→          items: cart.map((c) => ({ id: c.id, name: c.name, price: c.price, qty: c.qty })),
   179→          location: { city: location.city, pincode: location.pincode },
   180→          slot: deliverySlot,
   181→          coupon: coupon ?? undefined,
   182→          premiumWrap,
   183→        }),
   184→      });

   188→      setLastOrderId(data.orderId);
   189→      /* loyalty: stamp the card, maybe unlock the reward */

   191→        id: data.orderId,
   192→        total: data.total,
   193→        at: Date.now(),
   194→        items: count,
   195→      });
   196→      if (unlockedCode) {
   197→        setUnlocked(unlockedCode);
   198→        setTimeout(() => celebrationConfetti(), 650);

   200→          title: "Bloom Reward unlocked! 🌸",
   201→          description: `${unlockedCode} — ${resolveCoupon(unlockedCode)?.label ?? "a treat"} on your next order, on us.`,
   202→        });
   203→      }
   204→      celebrationConfetti();
   205→    } catch {
   206→      toast({ title: "Checkout hiccup 😢", description: "Please try again in a moment." });
   207→    } finally {
   208→      setCheckingOut(false);
   209→    }
   210→  };
   211→
   212→  const finish = () => {
   213→    clearCart();
   214→    setResult(null);
   215→    setUnlocked(null);
   216→    setPremiumWrap(false);
   217→    setCoupon(null);
   218→    setCouponInput("");
   219→    setCouponError(null);
   220→    setOpen(false);
   221→  };

   223→  return (
   224→    <AnimatePresence>
   225→      {isOpen && (

   236→
   237→          {/* Panel — bottom sheet on mobile (drag to dismiss) · slide-over on desktop */}

   245→            dragElastic={{ top: 0, bottom: 0.6 }}

   248→            }}
   249→            className={cn(
   250→              "absolute flex w-full flex-col bg-background shadow-2xl",
   251→              isMobile
   252→                ? "inset-x-0 bottom-0 max-h-[88dvh] rounded-t-[1.75rem]"
   253→                : "inset-y-0 right-0 max-w-md"
   254→            )}
   255→          >
   256→            {/* Header */}
   257→            <div className="border-b border-rose-100 bg-card dark:border-stone-800">
   258→              <div className="flex justify-center pt-2 md:hidden" aria-hidden>
   259→                <span className="h-1.5 w-12 rounded-full bg-stone-300 dark:bg-stone-600" />
   260→              </div>
   261→              <div className="flex items-center justify-between px-5 py-4">
   262→                <div className="flex items-center gap-2.5">
   263→                  <span className="grid h-9 w-9 place-items-center rounded-xl bg-brand-soft text-brand dark:bg-rose-950/50 dark:text-rose-300">
   264→                    <ShoppingBag className="h-4.5 w-4.5" aria-hidden />
   265→                  </span>
   266→                  <div>
   267→                    <h2 className="text-base font-extrabold text-foreground">Your Gift Bag</h2>

   270→                    </p>
   271→                  </div>
   272→                </div>
   273→                <button
   274→                  onClick={() => setOpen(false)}
   275→                  className="grid h-9 w-9 place-items-center rounded-full border border-stone-200 text-stone-500 transition hover:border-rose-300 hover:text-brand dark:border-stone-700"
   276→                  aria-label="Close gift bag"
   277→                >
   278→                  <X className="h-4.5 w-4.5" />
   279→                </button>
   280→              </div>
   281→            </div>
   282→
   283→            {result ? (
   284→              /* ---------- SUCCESS VIEW ---------- */
   285→              <div className="flex flex-1 flex-col items-center justify-center overflow-y-auto px-6 text-center scrollbar-slim">
   286→                <motion.div
   287→                  initial={{ scale: 0.6, opacity: 0 }}

   289→                  transition={{ type: "spring", stiffness: 260, damping: 20 }}
   290→                >
   291→                  <Lottie src={celebrationAnim} loop={false} className="h-44 w-44" />
   292→                </motion.div>
   293→                <motion.div
   294→                  initial={{ y: 24, opacity: 0 }}
   295→                  animate={{ y: 0, opacity: 1 }}
   296→                  transition={{ delay: 0.15 }}
   297→                >
   298→                  <h3 className="text-xl font-extrabold text-foreground">Order confirmed! 🎉</h3>
   299→                  <p className="mt-1 text-sm text-stone-500">
   300→                    Order <span className="font-mono font-bold text-brand">{result.orderId}</span> is
   301→                    being wrapped with love.
   302→                  </p>
   303→                  <div className="mt-4 w-full space-y-2 rounded-2xl border border-rose-100 bg-card p-4 text-left text-sm shadow-soft dark:border-stone-800">
   304→                    {unlocked && (
   305→                      <motion.div
   306→                        initial={{ opacity: 0, scale: 0.9 }}
   307→                        animate={{ opacity: 1, scale: 1 }}
   308→                        transition={{ delay: 0.5, type: "spring", stiffness: 300, damping: 18 }}
   309→                        className="flex flex-col gap-0.5 rounded-xl bg-gold-soft px-3 py-2 dark:bg-amber-950/40"
   310→                      >
   311→                        <span className="flex items-center gap-1.5 text-xs font-extrabold text-amber-700 dark:text-amber-300">

   313→                          Reward earned · {unlocked}
   314→                        </span>
   315→                        <span className="text-[11px] font-bold text-amber-700/90 dark:text-amber-300/90">
   316→                          {resolveCoupon(unlocked)?.label ?? "₹100 off"} on your next order
   317→                        </span>
   318→                      </motion.div>
   319→                    )}
   320→                    <div className="flex justify-between">
   321→                      <span className="text-stone-500">Paid</span>
   322→                      <span className="font-extrabold text-foreground">{formatINR(result.total)}</span>
   323→                    </div>
   324→                    {result.coupon && result.discount > 0 && (
   325→                      <div className="flex justify-between">

   327→                          <BadgeCheck className="h-3.5 w-3.5" aria-hidden />
   328→                          {result.coupon} saved
   329→                        </span>
   330→                        <span className="font-bold text-mint">−{formatINR(result.discount)}</span>
   331→                      </div>
   332→                    )}
   333→                    {result.coupon && result.freeShipping && (
   334→                      <div className="flex justify-between">
   335→                        <span className="flex items-center gap-1 text-mint">
   336→                          <BadgeCheck className="h-3.5 w-3.5" aria-hidden />
   337→                          {result.coupon} · free shipping
   338→                        </span>
   339→                        <span className="font-bold text-mint">−{formatINR(99)}</span>
   340→                      </div>

   342→                    <div className="flex justify-between">
   343→                      <span className="text-stone-500">Delivering to</span>
   344→                      <span className="font-bold text-foreground">{result.deliveryTo}</span>

   346→                    <div className="flex justify-between">

   350→                      </span>
   351→                    </div>
   352→                    <div className="flex justify-between">
   353→                      <span className="text-stone-500">Gift wrap</span>

   358→                    <div className="flex justify-between">
   359→                      <span className="text-stone-500">ETA</span>

   362→                  </div>
   363→                  <div className="mt-5 flex w-full gap-2">
   364→                    <button
   365→                      onClick={finish}
   366→                      className="flex flex-1 items-center justify-center gap-2 rounded-full bg-gradient-brand px-5 py-3 text-sm font-bold text-white shadow-lift transition hover:opacity-90"
   367→                    >
   368→                      Keep shopping <ArrowRight className="h-4 w-4" aria-hidden />

   372→                      whileTap={{ scale: 0.96 }}
   373→                      onClick={() => {
   374→                        setOpen(false);
   375→                        setTrackOpen(true);
   376→                      }}
   377→                      className="flex items-center justify-center gap-1.5 rounded-full border border-rose-200 bg-white px-4 py-3 text-sm font-bold text-foreground transition hover:border-brand hover:text-brand dark:border-stone-700 dark:bg-stone-900 dark:text-stone-200"
   378→                    >
   379→                      <Radar className="h-4 w-4 text-brand" aria-hidden />
   380→                      Track
   381→                    </motion.button>
   382→                  </div>
   383→                </motion.div>
   384→              </div>
   385→            ) : cart.length === 0 ? (
   386→              /* ---------- EMPTY VIEW ---------- */
   387→              <div className="flex flex-1 flex-col items-center justify-center gap-3 px-6 text-center">
   388→                <motion.div
   389→                  animate={{ y: [0, -10, 0], rotate: [0, -4, 4, 0] }}
   390→                  transition={{ repeat: Infinity, duration: 3.2 }}
   391→                  className="grid h-24 w-24 place-items-center rounded-full bg-brand-soft text-brand dark:bg-rose-950/50 dark:text-rose-300"
   392→                >
   393→                  <Gift className="h-10 w-10" aria-hidden />
   394→                </motion.div>
   395→                <h3 className="text-lg font-extrabold text-foreground">Your bag is feeling light</h3>
   396→                <p className="max-w-60 text-sm text-stone-500">
   397→                  Fill it with flowers, cakes & surprises — happiness ships free over{" "}
   398→                  {formatINR(FREE_SHIPPING_THRESHOLD)}.
   399→                </p>
   400→                <button
   401→                  onClick={() => setOpen(false)}
   402→                  className="mt-2 rounded-full bg-brand px-6 py-3 text-sm font-bold text-white shadow-lift transition hover:bg-rose-700"
   403→                >
   404→                  Browse bestsellers
   405→                </button>
   406→              </div>
   407→            ) : (
   408→              <>
   409→                {/* ---------- BLOOM REWARDS ---------- */}
   410→                <div className="border-b border-rose-100 bg-card px-5 py-3 dark:border-stone-800">
   411→                  <div className="flex items-center justify-between gap-3 rounded-2xl bg-gradient-to-r from-rose-50 via-pink-50 to-amber-50 px-3.5 py-2.5 ring-1 ring-rose-100 dark:from-rose-950/40 dark:via-stone-900 dark:to-amber-950/30 dark:ring-stone-700/80">
   412→                    <div className="flex items-center gap-2.5">
   413→                      <motion.span
   414→                        animate={{ rotate: [0, -8, 8, 0] }}
   415→                        transition={{ repeat: Infinity, duration: 3.4, ease: "easeInOut" }}
   416→                        className="grid h-8 w-8 shrink-0 place-items-center rounded-xl bg-brand text-white shadow-soft"
   417→                        aria-hidden
   418→                      >
   419→                        <Flower2 className="h-4 w-4" />
   420→                      </motion.span>
   421→                      <div>
   422→                        <p className="text-xs font-extrabold text-foreground">
   423→                          Bloom Rewards{" "}
   424→                          <span className="ml-0.5 rounded-full bg-brand-soft px-1.5 py-0.5 text-[9px] font-extrabold uppercase tracking-wide text-brand dark:bg-rose-950/50 dark:text-rose-300">
   425→                            {stamps}/{LOYALTY_TARGET} stamps
   426→                          </span>
   427→                        </p>
   428→                        <p className="text-[10px] font-semibold text-stone-400">
   429→                          {stamps > 0

   431→                            : `Every ${LOYALTY_TARGET}rd order earns ${nextRewardLabel}`}
   432→                        </p>
   433→                      </div>
   434→                    </div>
   435→                    <span className="flex items-center gap-1" aria-hidden>
   436→                      {Array.from({ length: LOYALTY_TARGET }).map((_, i) => (
   437→                        <motion.span
   438→                          key={i}
   439→                          initial={false}
   440→                          animate={
   441→                            i < stamps
   442→                              ? { scale: [1, 1.35, 1], rotate: [0, 18, 0] }
   443→                              : { scale: 1 }
   444→                          }
   445→                          transition={{ duration: 0.45, delay: i * 0.12 }}
   446→                        >
   447→                          <Flower2
   448→                            className={cn(
   449→                              "h-4.5 w-4.5",
   450→                              i < stamps ? "fill-brand text-brand" : "text-rose-200 dark:text-rose-800"
   451→                            )}
   452→                          />
   453→                        </motion.span>
   454→                      ))}
   455→                    </span>
   456→                  </div>
   457→                  {rewardCoupon && !coupon && (
   458→                    <motion.div
   459→                      initial={{ opacity: 0, y: 6 }}
   460→                      animate={{ opacity: 1, y: 0 }}
   461→                      className="mt-2 flex items-center justify-between rounded-2xl border border-dashed border-gold bg-gold-soft/70 px-3.5 py-2 dark:bg-amber-950/30"
   462→                    >
   463→                      <span className="flex items-center gap-1.5 text-[11px] font-extrabold text-amber-700 dark:text-amber-300">

   465→                        Your reward · {rewardCoupon}

   467→                      <span className="flex items-center gap-2">
   468→                        <button
   469→                          onClick={() => {
   470→                            setCoupon(rewardCoupon);
   471→                            setCouponInput(rewardCoupon);
   472→                            miniConfetti();
   473→                            toast({

   478→                          className="rounded-full bg-charcoal px-3 py-1 text-[10px] font-extrabold text-cream transition hover:bg-stone-700"
   479→                        >
   480→                          Apply
   481→                        </button>
   482→                        <button
   483→                          onClick={dismissReward}
   484→                          aria-label="Dismiss reward"
   485→                          className="text-stone-400 transition hover:text-brand"

   489→                      </span>
